# How far terrain loads and whether it holds still: shots while it comes in, then frozen ones a few frames apart.
cmd gamerule sendCommandFeedback false
cmd gamerule doDaylightCycle false
cmd time set 6000
cmd gamemode spectator @a
resize 2560 1369
cmd tp @a 0 140 0 135 20
wait 100
shot fk_load1
wait 100
shot fk_load2
wait 100
shot fk_load3
settle
wait 200
cmd tick freeze
wait 60
shot fk_a
wait 3
shot fk_b
wait 3
shot fk_c
wait 3
shot fk_d
dump fk
wait 10
cmd tick unfreeze
error
