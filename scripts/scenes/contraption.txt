# Mechanical pistons that assemble and move a contraption, seen from close by, some pictures apart. Poles behind the piston give
# it its reach; the blocks in front of it ride along. Two of them, turned in opposite directions: one of them has to extend.
cmd gamerule sendCommandFeedback false
cmd gamemode creative @a
cmd clear @a
cmd fill -8 99 -2 8 99 20 minecraft:smooth_stone
cmd fill -8 100 -2 8 110 20 minecraft:air
cmd setblock -2 100 2 create:piston_extension_pole[facing=south]
cmd setblock -2 100 3 create:piston_extension_pole[facing=south]
cmd setblock -2 100 4 create:piston_extension_pole[facing=south]
cmd setblock -2 100 5 create:mechanical_piston[facing=south,axis_along_first=true,state=retracted]
cmd setblock -2 100 6 minecraft:oak_planks
cmd setblock 2 100 2 create:piston_extension_pole[facing=south]
cmd setblock 2 100 3 create:piston_extension_pole[facing=south]
cmd setblock 2 100 4 create:piston_extension_pole[facing=south]
cmd setblock 2 100 5 create:mechanical_piston[facing=south,axis_along_first=true,state=retracted]
cmd setblock 2 100 6 minecraft:oak_planks
gui true
slot 0
cmd tp @a 0 106 -1 0 40
settle
shot p0_no_motor
cmd setblock -3 100 5 create:creative_motor[facing=east]
cmd setblock 3 100 5 create:creative_motor[facing=west]
wait 25
shot p1_moving
wait 25
shot p2_moving
wait 25
shot p3_moving
wait 60
shot p4_moving
error
